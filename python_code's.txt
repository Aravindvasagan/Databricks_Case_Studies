92. Find duplicates
def duplicates(arr):
    seen = set()
    dup = set()
    for x in arr:
        if x in seen:
            dup.add(x)
        seen.add(x)
    return list(dup)

def flatten(lst):
    res = []
    for i in lst:
        if isinstance(i, list):
            res.extend(flatten(i))
        else:
            res.append(i)
    return res

def first_unique(s):
    count = Counter(s)
    for c in s:
        if count[c] == 1:
            return c

def two_sum(nums, target):
    m = {}   # hashmap

    for i, n in enumerate(nums):
        diff = target - n

        if diff in m:
            return [m[diff], i]

        m[n] = i

🔹 Simple Flow Example
def withdraw(balance, amount):
    try:
        if amount > balance:
            raise ValueError("Insufficient balance")
        
        assert amount > 0, "Amount must be positive"
        
        return balance - amount

    except ValueError as e:
        print("Error:", e)
